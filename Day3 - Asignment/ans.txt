# Column names

# Missing values

# =========================
# Q6 - CORRELATION HEATMAP
# =========================

columns = [
    "Age",
    "Gene_A_Expression",
    "Gene_B_Expression",
    "Gene_C_Expression",
    "Blood_Pressure",
    "Glucose"
]

corr = df[columns].corr()

sns.heatmap(
    corr,

)

plt.title("Correlation Heatmap")

# Find strongest positive correlation

for i in range(len(corr2)):
    corr2.iloc[i, i] = 0

print("Strongest positive correlation:")

print(corr2.stack().idxmax())
print(corr2.stack().max())


# =========================
# Q7 - VOLCANO PLOT
# =========================

genes = [
    "Gene_A_Expression",
    "Gene_B_Expression",
    "Gene_C_Expression"
]

disease = df[df["Disease_Status"] == "Disease"]
healthy = df[df["Disease_Status"] == "Healthy"]

results = []

for gene in genes:

    disease_mean = disease[gene].mean()
    healthy_mean = healthy[gene].mean()

    log2fc = np.log2(disease_mean / healthy_mean)

    p = ttest_ind(
        disease[gene].dropna(),
        healthy[gene].dropna()
    ).pvalue

    results.append([
        gene,
        disease_mean,
        healthy_mean,
        log2fc,
        p
    ])

result = pd.DataFrame(
    results,
    columns=[
        "Gene",
        "Disease Mean",
        "Healthy Mean",
        "Log2FC",
        "P-value"
    ]
)

print(result)


# Volcano plot

result["-log10(p)"] = -np.log10(result["P-value"])

plt.scatter(
    result["Log2FC"],
    result["-log10(p)"]
)

for i in range(len(result)):
    plt.text(
        result["Log2FC"][i],
        result["-log10(p)"][i],
        result["Gene"][i]
    )

plt.axvline(0.5, color="red", linestyle="--")
plt.axvline(-0.5, color="red", linestyle="--")
plt.axhline(-np.log10(0.05), color="blue", linestyle="--")

plt.xlabel("Log2 Fold Change")
plt.ylabel("-Log10(P-value)")
plt.title("Volcano Plot")
plt.show()

# Significant genes
print("Significant genes:")

print(
    result[
        (abs(result["Log2FC"]) > 0.5) &
        (result["P-value"] < 0.05)
    ]["Gene"]
)


# =========================
# Q8 - UPSET PLOT
# =========================

mutation = df[
    ["Mutation_A", "Mutation_B", "Mutation_C"]
].replace({
    "Yes": True,
    "No": False
})

print(mutation)

# Create UpSet plot
data = from_indicators(
    mutation.columns,
    mutation
)

UpSet(
    data,
    show_counts=True
).plot()

plt.show()

# Most common combination
print("Most common mutation combination:")

print(mutation.value_counts().head(1))
